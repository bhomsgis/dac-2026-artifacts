// Check that when ptw_arb is IDLE and a request is waiting, ptw_itlb_comm_o.ptw_ready and ptw_dtlb_comm_o.ptw_ready are 0 and next_state is SERVING_iTLB in the next cycle
as__IdleReadyWaiting: assert property ((ptw_arb.current_state == ptw_arb.IDLE && ptw_arb.is_req_waiting_q) |=> (!ptw_arb.ptw_itlb_comm_o.ptw_ready && !ptw_arb.ptw_dtlb_comm_o.ptw_ready && ptw_arb.next_state == ptw_arb.SERVING_iTLB));
